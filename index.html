async function send(){ const input=document.getElementById('question'); const q=input.value.trim(); if(!q)return; input.value=''; document.getElementById('information').textContent = 'PROCESSING COMMAND...'; reply.textContent = 'JAANI IS THINKING...'; try { const response = await fetch('http://localhost:8000/chat', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ message: q }) }); const data = await response.json(); document.getElementById('information').textContent = 'USER COMMAND: ' + q + ' | CONNECTED'; answer(data.reply); } catch(err) { reply.textContent = 'CONNECTION ERROR WITH AI BRAIN'; setMode('idle'); } }
